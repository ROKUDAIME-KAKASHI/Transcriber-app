## Phase Goal

**As a** meeting recorder using the Android app or the website, **I want to** export and share any meeting in one tap and always find my way back, **so that** I can get the minutes or transcript to people right after the meeting without getting stuck.

<objective>
Vertical slice 1 (navigation + platform detection): the user can always find their way back. Brand link goes to Record from anywhere, the meeting page has a reliable back arrow, and in the new APK the Android back key closes overlays, goes back, and minimizes (not exits) while recording. Along the way this plan installs every native plugin and export dependency for the milestone (one APK rebuild, D-09) and creates the single `src/lib/native.ts` capability layer (D-08) that every later plan uses.

Purpose: NAV-01/02/03 and the SHARE-05 detection rule (`hasPlugin = isNativePlatform() && isPluginAvailable(name)`).
Output: navStack, overlays, recordingFlag, native.ts, NavTracker, NativeBackHandler, updated AppShell/MeetingView header/Providers/Recorder, pinned dependencies.
</objective>

// src/lib/native.ts
export type PluginName = "Share" | "Filesystem" | "App" | "Network" | "Haptics" | "DocumentExport" | "MeetingRecorder";
export function isNative(): boolean;                 // typeof window !== "undefined" && Capacitor.isNativePlatform()
export function hasPlugin(name: PluginName): boolean; // isNative() && Capacitor.isPluginAvailable(name)
export function getShare(): Promise<typeof import("@capacitor/share").Share>;
export function getFilesystem(): Promise<{ Filesystem: ...; Directory: ... }>;
export function getApp(): Promise<typeof import("@capacitor/app").App>;
export function getNetwork(): Promise<typeof import("@capacitor/network").Network>;
export function getHaptics(): Promise<typeof import("@capacitor/haptics").Haptics>;
export interface DocumentExportPlugin {
  htmlToPdf(o: { html: string; fileName: string }): Promise<{ uri: string }>;
  printHtml(o: { html: string; jobName: string }): Promise<{ started: boolean }>;
}
// rejection codes (PluginCall.reject code): "BUSY" | "CANCELLED" | "LAYOUT_FAILED" | "WRITE_FAILED" | "IO" | "TIMEOUT" | "SHIM_FAILED" | "BAD_ARGS"
export const DocumentExport: DocumentExportPlugin;   // registerPlugin("DocumentExport"), no web impl
export const EXPORTS_DIR = "exports";               // Directory.Cache/exports, shared with the Java plugin (getCacheDir()/exports)
export function cleanupOldExports(): Promise<void>; // native + Filesystem only, deletes files older than 86_400_000 ms, never throws

// src/lib/overlays.ts
export function pushOverlay(close: () => void): () => void; // returns unregister
export function closeTopOverlay(): boolean;                 // true if something was closed
</interfaces>
